核心代码

Node* merge(Node* A, Node* B) {
    Node dummy, *tail = &dummy;
    dummy.next = NULL;
    while (A && B) {
        if (A->data <= B->data) { tail->next = A; A = A->next; }
        else { tail->next = B; B = B->next; }
        tail = tail->next;
    }
    tail->next = A ? A : B;
    return dummy.next;
}
